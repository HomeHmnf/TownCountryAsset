===== TARGET: STATISTICS IMPORTER =====
"""Transactional importer for normalized WP1.5 provider observations.

Provider-specific parsers must produce the normalized JSON contract documented
in docs/data-sources/wp1-5-sources.md. This module never parses raw provider
files and never infers country identities or missing values.
"""
from __future__ import annotations

import hashlib
import json
import math
import sqlite3
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from tools.database.connection import DEFAULT_DATABASE_PATH, REPOSITORY_ROOT, connect_database
from tools.database.observation_contract import (
    ALLOWED_VALUE_STATUSES, validate_observation_value,
)
from tools.database.application_approval import (
    ApplicationApprovalError,
    build_application_artifact,
    require_approved_application,
    record_application_applied,
)
from tools.database.schema import apply_migrations, validate_database
from tools.database.source_crosswalk import CrosswalkError, load_crosswalk
from tools.sources.registry import SourceError, canonical_source_id, load_registry

DEFAULT_CONFIG = Path(__file__).resolve().parents[2] / "config" / "wp1_5_data_sources.json"


class StatisticsImportError(RuntimeError):
    """Raised when a normalized WP1.5 dataset cannot be safely imported."""


def _read_object(path: str | Path, label: str) -> dict[str, Any]:
    try:
        value = json.loads(Path(path).read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as exc:
        raise StatisticsImportError(f"Cannot read {label} {path}: {exc}") from exc
    if not isinstance(value, dict):
        raise StatisticsImportError(f"{label} must be a JSON object")
    return value


def _canonical_json(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False)


def _utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z")


def _parse_aware_timestamp(value: str, label: str) -> datetime:
    try:
        parsed = datetime.fromisoformat(value.replace("Z", "+00:00"))
    except ValueError as exc:
        raise StatisticsImportError(f"{label} must be an ISO-8601 timestamp") from exc
    if parsed.tzinfo is None or parsed.utcoffset() is None:
        raise StatisticsImportError(f"{label} must include a timezone")
    return parsed


def _text(value: Any, label: str) -> str:
    if not isinstance(value, str) or not value.strip():
        raise StatisticsImportError(f"{label} must be a non-empty string")
    return value


def _validate_manifest(manifest: dict[str, Any], source_key: str) -> tuple[str, str, str, str]:
    if manifest.get("manifest_schema_version") != 1 or manifest.get("work_package") != "WP1.5":
        raise StatisticsImportError("Unsupported WP1.5 source manifest")
    if manifest.get("source_key") != source_key:
        raise StatisticsImportError("Source manifest source_key does not match normalized dataset")
    release = _text(manifest.get("release"), "manifest.release")
    sha256 = _text(manifest.get("sha256"), "manifest.sha256")
    if len(sha256) != 64 or any(c not in "0123456789abcdef" for c in sha256):
        raise StatisticsImportError("manifest.sha256 must be a lowercase SHA-256 digest")
    raw_path_value = _text(manifest.get("local_raw_path"), "manifest.local_raw_path")
    raw_file = Path(raw_path_value)
    if not raw_file.is_absolute():
        raw_file = (REPOSITORY_ROOT / raw_file).resolve()
    else:
        raw_file = raw_file.resolve()
    try:
        relative_raw_path = raw_file.relative_to(REPOSITORY_ROOT.resolve())
    except ValueError as exc:
        raise StatisticsImportError("Captured raw source file must be inside the repository") from exc
    if not relative_raw_path.parts or relative_raw_path.parts[0] != "raw_data":
        raise StatisticsImportError("Captured raw source file must be directly under the repository raw_data tree")
    try:
        raw_bytes = raw_file.read_bytes()
    except OSError as exc:
        raise StatisticsImportError(f"Cannot read captured raw source file: {raw_file}") from exc
    actual_sha = hashlib.sha256(raw_bytes).hexdigest()
    if actual_sha != sha256:
        raise StatisticsImportError("Captured raw source checksum does not match its manifest")
    if not raw_bytes or manifest.get("byte_count") != len(raw_bytes):
        raise StatisticsImportError("Captured raw source byte count does not match its manifest")
    source_url = _text(manifest.get("final_url") or manifest.get("source_url"), "manifest source URL")
    if not source_url.startswith("https://"):
        raise StatisticsImportError("Source manifest URL must use HTTPS")
    retrieved_at = _text(manifest.get("retrieved_at"), "manifest.retrieved_at")
    try:
        retrieved_timestamp = datetime.fromisoformat(retrieved_at.replace("Z", "+00:00"))
    except ValueError as exc:
        raise StatisticsImportError("manifest.retrieved_at must be an ISO-8601 timestamp") from exc
    if retrieved_timestamp.tzinfo is None or retrieved_timestamp.utcoffset() is None:
        raise StatisticsImportError("manifest.retrieved_at must include a timezone")
    additional_inputs = manifest.get("additional_inputs", [])
    if not isinstance(additional_inputs, list):
        raise StatisticsImportError("manifest.additional_inputs must be an array")
    for index, item in enumerate(additional_inputs):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.additional_inputs[{index}] must be an object")
        additional_path_value = _text(item.get("local_raw_path"), f"manifest.additional_inputs[{index}].local_raw_path")
        additional_path = Path(additional_path_value)
        if not additional_path.is_absolute():
            additional_path = (REPOSITORY_ROOT / additional_path).resolve()
        else:
            additional_path = additional_path.resolve()
        try:
            additional_relative = additional_path.relative_to(REPOSITORY_ROOT.resolve())
        except ValueError as exc:
            raise StatisticsImportError("Additional source snapshots must be inside the repository") from exc
        if not additional_relative.parts or additional_relative.parts[0] != "raw_data":
            raise StatisticsImportError("Additional source snapshots must be under the repository raw_data tree")
        try:
            additional_bytes = additional_path.read_bytes()
        except OSError as exc:
            raise StatisticsImportError(f"Cannot read additional source snapshot: {additional_path}") from exc
        additional_sha = _text(item.get("sha256"), f"manifest.additional_inputs[{index}].sha256")
        if len(additional_sha) != 64 or any(c not in "0123456789abcdef" for c in additional_sha):
            raise StatisticsImportError(f"manifest.additional_inputs[{index}].sha256 is invalid")
        if hashlib.sha256(additional_bytes).hexdigest() != additional_sha:
            raise StatisticsImportError(f"Additional source snapshot checksum mismatch: {additional_path}")
        if item.get("byte_count") != len(additional_bytes):
            raise StatisticsImportError(f"Additional source snapshot byte count mismatch: {additional_path}")
    upstream_manifests = manifest.get("upstream_manifests", [])
    if not isinstance(upstream_manifests, list):
        raise StatisticsImportError("manifest.upstream_manifests must be an array")
    for index, item in enumerate(upstream_manifests):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.upstream_manifests[{index}] must be an object")
        manifest_path_value = _text(item.get("path"), f"manifest.upstream_manifests[{index}].path")
        upstream_path = Path(manifest_path_value)
        if not upstream_path.is_absolute():
            upstream_path = (REPOSITORY_ROOT / upstream_path).resolve()
        else:
            upstream_path = upstream_path.resolve()
        try:
            upstream_relative = upstream_path.relative_to(REPOSITORY_ROOT.resolve())
        except ValueError as exc:
            raise StatisticsImportError("Upstream acquisition manifests must be inside the repository") from exc
        if not upstream_path.is_file():
            raise StatisticsImportError(f"Upstream acquisition manifest is missing: {upstream_path}")
        if not upstream_relative.parts:
            raise StatisticsImportError("Invalid upstream acquisition manifest path")
        try:
            upstream_digest = hashlib.sha256(upstream_path.read_bytes()).hexdigest()
        except OSError as exc:
            raise StatisticsImportError(f"Cannot read upstream acquisition manifest: {upstream_path}") from exc
        expected_digest = _text(
            item.get("sha256"), f"manifest.upstream_manifests[{index}].sha256"
        )
        if upstream_digest != expected_digest:
            raise StatisticsImportError(f"Upstream acquisition manifest checksum mismatch: {upstream_path}")
    for optional_key in ("attribution", "license_status"):
        optional_value = manifest.get(optional_key)
        if optional_value is not None and not isinstance(optional_value, str):
            raise StatisticsImportError(f"manifest.{optional_key} must be text when supplied")
    terms_url = manifest.get("terms_url")
    if terms_url is not None and (not isinstance(terms_url, str) or not terms_url.startswith("https://")):
        raise StatisticsImportError("manifest.terms_url must be HTTPS when supplied")
    return release, sha256, source_url, retrieved_at


def _validate_observation(
    row: Any,
    *,
    index: int,
    field_by_id: dict[int, dict[str, Any]],
    source_key: str,
    crosswalk_by_game: dict[str, str],
) -> dict[str, Any]:
    label = f"observations[{index}]"
    if not isinstance(row, dict):
        raise StatisticsImportError(f"{label} must be an object")
    game_id = _text(row.get("game_id"), f"{label}.game_id")
    source_id = _text(row.get("source_id"), f"{label}.source_id")
    if crosswalk_by_game.get(game_id) != source_id:
        raise StatisticsImportError(f"{label} game_id/source_id pair is absent from the explicit crosswalk")
    field_id = row.get("field_id")
    if isinstance(field_id, bool) or not isinstance(field_id, int) or field_id not in field_by_id:
        raise StatisticsImportError(f"{label}.field_id is not in the approved 41-field contract")
    field = field_by_id[field_id]
    if field["primary_source"] != source_key:
        raise StatisticsImportError(
            f"{label} source {source_key!r} is not the contracted primary source for field {field_id}"
        )
    period = _text(row.get("observation_period"), f"{label}.observation_period")
    status = row.get("value_status")
    if not isinstance(status, str) or status not in ALLOWED_VALUE_STATUSES:
        raise StatisticsImportError(f"{label}.value_status is invalid")
    if "value" not in row:
        raise StatisticsImportError(f"{label}.value must be present; use explicit JSON null only for unavailable/not_applicable")
    value = row.get("value")
    unit = row.get("unit")
    if not isinstance(unit, str) or not unit.strip():
        raise StatisticsImportError(f"{label}.unit must be a non-empty string")
    contracted_unit = field.get("unit")
    if not isinstance(contracted_unit, str):
        raise StatisticsImportError(f"Field contract unit is invalid for field {field_id}")
    allowed_units = {contracted_unit, *(part.strip() for part in contracted_unit.split("_or_") if part.strip())}
    if unit not in allowed_units:
        raise StatisticsImportError(
            f"{label}.unit {unit!r} is not permitted by field contract {contracted_unit!r}"
        )
    if field["storage"] == "scalar":
        if value is None:
            if status not in {"unavailable", "not_applicable"}:
                raise StatisticsImportError(f"{label} NULL value requires unavailable/not_applicable status")
        else:
            if isinstance(value, bool) or not isinstance(value, (int, float)):
                raise StatisticsImportError(f"{label} scalar value must be a finite number or explicit NULL")
            try:
                finite = math.isfinite(float(value))
            except (OverflowError, ValueError):
                finite = False
            if not finite:
                raise StatisticsImportError(f"{label} scalar value must be a finite number or explicit NULL")
            value = float(value)
            if not validate_observation_value(value, status, "scalar"):
                raise StatisticsImportError(
                    f"{label} populated scalar has an invalid value/status combination"
                )
    else:
        if value is None or not isinstance(value, (dict, list)):
            raise StatisticsImportError(f"{label} structured value must be a JSON object or array")
        if not validate_observation_value(value, status, "structured"):
            raise StatisticsImportError(
                f"{label} structured value requires a valid observed/estimated/projected status"
            )
        try:
            _canonical_json(value)
        except (TypeError, ValueError) as exc:
            raise StatisticsImportError(f"{label} structured value is not valid finite JSON") from exc
    source_reference = row.get("source_reference")
    if source_reference is not None and (
        not isinstance(source_reference, str) or not source_reference.strip()
    ):
        raise StatisticsImportError(f"{label}.source_reference must be non-empty text when supplied")
    methodology = row.get("methodology", field.get("method"))
    if not isinstance(methodology, str) or not methodology.strip():
        raise StatisticsImportError(f"{label}.methodology must be non-empty text")
    return {
        "game_id": game_id,
        "source_id": source_id,
        "field": field,
        "value": value,
        "unit": unit,
        "status": status,
        "period": period,
        "source_reference": source_reference,
        "methodology": methodology,
    }


def build_wp1_5_application_artifact(
    *,
    observations_path: str | Path,
    crosswalk_path: str | Path,
    manifest_path: str | Path,
    config_path: str | Path = DEFAULT_CONFIG,
) -> dict[str, Any]:
    """Fingerprint the complete WP1.5 input set that an approver reviewed."""
    manifest = _read_object(manifest_path, "source manifest")
    normalized = _read_object(observations_path, "normalized observations")
    source_key = _text(normalized.get("source_key"), "normalized.source_key")
    _validate_manifest(manifest, source_key)
    if normalized.get("release") != manifest.get("release"):
        raise StatisticsImportError("Normalized observations and manifest release do not match")
    if normalized.get("source_sha256") != manifest.get("sha256"):
        raise StatisticsImportError("Normalized observations and manifest checksum do not match")
    raw_path = Path(_text(manifest.get("local_raw_path"), "manifest.local_raw_path"))
    if not raw_path.is_absolute():
        raw_path = (REPOSITORY_ROOT / raw_path).resolve()
    inputs: list[Path] = [
        Path(observations_path), Path(crosswalk_path), Path(manifest_path),
        Path(config_path), raw_path,
    ]
    additional_inputs = manifest.get("additional_inputs", [])
    if not isinstance(additional_inputs, list):
        raise StatisticsImportError("manifest.additional_inputs must be an array")
    for index, item in enumerate(additional_inputs):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.additional_inputs[{index}] must be an object")
        additional_path = Path(_text(
            item.get("local_raw_path"),
            f"manifest.additional_inputs[{index}].local_raw_path",
        ))
        inputs.append(additional_path)
    upstream_manifests = manifest.get("upstream_manifests", [])
    if not isinstance(upstream_manifests, list):
        raise StatisticsImportError("manifest.upstream_manifests must be an array")
    for index, item in enumerate(upstream_manifests):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.upstream_manifests[{index}] must be an object")
        inputs.append(Path(_text(
            item.get("path"), f"manifest.upstream_manifests[{index}].path"
        )))
    selection_policy = REPOSITORY_ROOT / "config" / "source_selection.json"
    policy_version = None
    if selection_policy.is_file():
        inputs.append(selection_policy)
        policy = _read_object(selection_policy, "source-selection policy")
        policy_version = policy.get("policy_version")
    return build_application_artifact(
        domain="wp1_5_statistics",
        inputs=inputs,
        metadata={
            "source_key": normalized.get("source_key"),
            "release": normalized.get("release"),
            "source_sha256": normalized.get("source_sha256"),
            "policy_version": policy_version,
        },
    )


def import_wp1_5_dataset(
    *,
    observations_path: str | Path,
    crosswalk_path: str | Path,
    manifest_path: str | Path,
    database_path: str | Path = DEFAULT_DATABASE_PATH,
    config_path: str | Path = DEFAULT_CONFIG,
    approval_id: str,
) -> dict[str, Any]:
    """Import one approved normalized provider snapshot transactionally."""
    normalized = _read_object(observations_path, "normalized observations")
    manifest = _read_object(manifest_path, "source manifest")
    config = _read_object(config_path, "WP1.5 field contract")
    if normalized.get("schema_version") != 1 or normalized.get("work_package") != "WP1.5":
        raise StatisticsImportError("Unsupported normalized observation schema")
    source_key = _text(normalized.get("source_key"), "normalized.source_key")
    registry_reference = _text(config.get("source_registry"), "WP1.5 contract source_registry")
    registry_path = Path(registry_reference)
    if not registry_path.is_absolute():
        # Registry references in project configuration are repository-root-relative,
        # not relative to the directory containing the referencing config file.
        registry_path = (REPOSITORY_ROOT / registry_path).resolve()
    try:
        source_registry = load_registry(registry_path)
    except SourceError as exc:
        raise StatisticsImportError(f"Cannot load project source registry: {exc}") from exc
    canonical_source_key = canonical_source_id(source_registry, source_key)
    if canonical_source_key is None:
        raise StatisticsImportError(f"Unknown source ID in project source registry: {source_key}")
    # Keep the historical ID in manifests, crosswalks and database provenance;
    # use its canonical provider ID only for current field-contract validation.
    release, source_sha, source_url, retrieved_at = _validate_manifest(manifest, source_key)
    if normalized.get("source_key") != manifest.get("source_key"):
        raise StatisticsImportError("Normalized observations and source manifest do not match")
    if normalized.get("source_sha256") != source_sha or normalized.get("release") != release:
        raise StatisticsImportError("Normalized observations do not identify the captured source release/checksum")

    fields = config.get("fields")
    if not isinstance(fields, list):
        raise StatisticsImportError("WP1.5 contract fields must be an array")
    field_by_id = {item.get("id"): item for item in fields if isinstance(item, dict)}
    if len(field_by_id) != 41:
        raise StatisticsImportError("WP1.5 contract must enumerate exactly 41 distinct fields")

    try:
        crosswalk = load_crosswalk(
            crosswalk_path,
            expected_dataset=source_key,
        )
    except CrosswalkError as exc:
        raise StatisticsImportError(f"Invalid source crosswalk: {exc}") from exc
    if crosswalk.source_key != source_key:
        raise StatisticsImportError("Crosswalk source_key does not match source contract")
    crosswalk_by_game = {item.game_id: item.source_id for item in crosswalk.mappings}

    rows = normalized.get("observations")
    if not isinstance(rows, list) or not rows:
        raise StatisticsImportError("Normalized observations must be a non-empty array")
    validated = [
        _validate_observation(
            row,
            index=index,
            field_by_id=field_by_id,
            source_key=canonical_source_key,
            crosswalk_by_game=crosswalk_by_game,
        )
        for index, row in enumerate(rows)
    ]
    keys = [(row["game_id"], row["field"]["id"]) for row in validated]
    if len(keys) != len(set(keys)):
        raise StatisticsImportError("Duplicate country/field observations in normalized dataset")

    try:
        application_artifact = build_wp1_5_application_artifact(
            observations_path=observations_path,
            crosswalk_path=crosswalk_path,
            manifest_path=manifest_path,
            config_path=config_path,
        )
    except (ApplicationApprovalError, OSError, ValueError) as exc:
        raise StatisticsImportError(f"Cannot fingerprint WP1.5 application inputs: {exc}") from exc

    # Ensure the in-memory payloads validated above are exactly the bytes
    # represented by the approval artifact, not a concurrent file revision.
    if _read_object(observations_path, "normalized observations") != normalized:
        raise StatisticsImportError("Normalized observations changed while preparing the approved application")
    if _read_object(manifest_path, "source manifest") != manifest:
        raise StatisticsImportError("Source manifest changed while preparing the approved application")
    if _read_object(config_path, "WP1.5 field contract") != config:
        raise StatisticsImportError("WP1.5 field contract changed while preparing the approved application")
    _validate_manifest(manifest, source_key)

    connection = connect_database(database_path)
    try:
        apply_migrations(connection)
        known = {
            str(row["game_id"])
            for row in connection.execute("SELECT game_id FROM countries").fetchall()
        }
        crosswalk_unknown = sorted({item.game_id for item in crosswalk.mappings} - known)
        if crosswalk_unknown:
            raise StatisticsImportError(
                "Crosswalk references game_id values absent from canonical database: "
                + ", ".join(crosswalk_unknown[:10])
            )
        unknown = sorted({row["game_id"] for row in validated} - known)
        if unknown:
            raise StatisticsImportError(
                "Normalized observations reference game_id values absent from canonical database: "
                + ", ".join(unknown[:10])
            )
        connection.execute("BEGIN IMMEDIATE")
        try:
            require_approved_application(
                connection, approval_id=approval_id, artifact=application_artifact
            )
        except ApplicationApprovalError as exc:
            status_row = connection.execute(
                "SELECT status FROM application_approvals WHERE approval_id = ?",
                (approval_id,),
            ).fetchone()
            if status_row is not None and status_row["status"] == "invalidated":
                # Persist invalidation even though no domain data was written.
                connection.commit()
            elif connection.in_transaction:
                connection.rollback()
            raise StatisticsImportError(str(exc)) from exc
        imported_at = _utc_now()
        existing_imports = connection.execute(
            """SELECT import_id, source_version, source_url, retrieval_date,
                      attribution, license, import_status
               FROM dataset_imports WHERE dataset = ? AND sha256 = ? ORDER BY import_id""",
            (source_key, source_sha),
        ).fetchall()
        if len(existing_imports) > 1:
            raise StatisticsImportError("Multiple dataset_imports rows exist for this source/checksum")
        existing_import = existing_imports[0] if existing_imports else None
        if existing_import is None:
            cursor = connection.execute(
                """INSERT INTO dataset_imports
                   (dataset, source_version, source_url, retrieval_date, sha256,
                    attribution, license, import_status, imported_at)
                   VALUES (?, ?, ?, ?, ?, ?, ?, 'candidate', ?)""",
                (
                    source_key, release, source_url, retrieved_at, source_sha,
                    manifest.get("attribution"), manifest.get("license_status"), imported_at,
                ),
            )
            import_id = int(cursor.lastrowid)
        else:
            import_id = int(existing_import["import_id"])
            expected = (
                release, source_url, retrieved_at, manifest.get("attribution"),
                manifest.get("license_status"), "candidate",
            )
            actual = (
                existing_import["source_version"], existing_import["source_url"],
                existing_import["retrieval_date"], existing_import["attribution"],
                existing_import["license"], existing_import["import_status"],
            )
            if actual != expected:
                raise StatisticsImportError("Existing dataset import metadata conflicts with source manifest")

            # A checksum identifies immutable source bytes. Replaying those bytes
            # must reproduce the exact current rows from that import; it must not
            # append observations omitted by a changed normalizer or resurrect
            # rows that a newer release has already superseded.
            for mapping in crosswalk.mappings:
                existing_mapping = connection.execute(
                    """SELECT game_id, is_current FROM country_sources
                       WHERE dataset = ? AND source_id = ?""",
                    (source_key, mapping.source_id),
                ).fetchone()
                if (
                    existing_mapping is None
                    or existing_mapping["game_id"] != mapping.game_id
                    or existing_mapping["is_current"] != 1
                ):
                    raise StatisticsImportError(
                        f"Existing import crosswalk no longer matches source mapping {mapping.source_id}"
                    )

            scalar_count = 0
            structured_count = 0
            for row in validated:
                field = row["field"]
                reference = row["source_reference"] or source_url
                if field["storage"] == "scalar":
                    expected_values = (
                        row["value"], row["unit"], None, row["period"], row["status"],
                        reference, release, row["methodology"], retrieved_at, import_id,
                    )
                    existing = connection.execute(
                        """SELECT value, unit, observed_at, observation_period, value_status,
                                  source_reference, source_version, methodology, retrieved_at, import_id
                           FROM current_statistics WHERE game_id = ? AND statistic_key = ?""",
                        (row["game_id"], field["key"]),
                    ).fetchone()
                    if existing is None or tuple(existing) != expected_values:
                        raise StatisticsImportError(
                            f"Refusing non-newer or altered replay of source release for {row['game_id']}/{field['key']}"
                        )
                    scalar_count += 1
                else:
                    expected_values = (
                        _canonical_json({"value": row["value"], "unit": row["unit"]}),
                        row["status"], row["period"], reference, release,
                        row["methodology"], retrieved_at, import_id,
                    )
                    existing = connection.execute(
                        """SELECT payload_json, value_status, observation_period, source_reference,
                                  source_version, methodology, retrieved_at, import_id
                           FROM selected_game_data WHERE game_id = ? AND data_category = 'wp1_5'
                             AND data_key = ?""",
                        (row["game_id"], field["key"]),
                    ).fetchone()
                    if existing is None or tuple(existing) != expected_values:
                        raise StatisticsImportError(
                            f"Refusing non-newer or altered replay of source release for {row['game_id']}/{field['key']}"
                        )
                    structured_count += 1

            stored_scalar = connection.execute(
                "SELECT COUNT(*) FROM current_statistics WHERE import_id = ?", (import_id,)
            ).fetchone()[0]
            stored_structured = connection.execute(
                "SELECT COUNT(*) FROM selected_game_data WHERE import_id = ?", (import_id,)
            ).fetchone()[0]
            if stored_scalar != scalar_count or stored_structured != structured_count:
                raise StatisticsImportError(
                    "Refusing replay because normalized observations do not exactly match the original import"
                )
            issues = validate_database(connection)
            if issues:
                raise StatisticsImportError("Database validation failed: " + "; ".join(issues))
            try:
                record_application_applied(
                    connection, approval_id=approval_id,
                    application_id=f"wp1-5:{source_key}:{release}:{source_sha[:12]}",
                    details={"idempotent": True, "import_id": import_id},
                )
            except ApplicationApprovalError as exc:
                raise StatisticsImportError(str(exc)) from exc
            connection.commit()
            return {
                "dataset": source_key,
                "release": release,
                "import_id": import_id,
                "observations": len(validated),
                "scalar_observations": scalar_count,
                "structured_observations": structured_count,
                "crosswalk_mappings": len(crosswalk.mappings),
                "integrity": "ok",
                "foreign_keys": "ok",
                "idempotent": True,
            }

        # Persist explicit provider IDs as source provenance.
        # A changed mapping requires a separate identity-review operation.
        for mapping in crosswalk.mappings:
            existing_mapping = connection.execute(
                """SELECT game_id, is_current FROM country_sources
                   WHERE dataset = ? AND source_id = ?""",
                (source_key, mapping.source_id),
            ).fetchone()
            if existing_mapping is None:
                connection.execute(
                    """INSERT INTO country_sources
                       (game_id, dataset, source_id, is_current, import_id)
                       VALUES (?, ?, ?, 1, ?)""",
                    (mapping.game_id, source_key, mapping.source_id, import_id),
                )
            elif existing_mapping["game_id"] != mapping.game_id or existing_mapping["is_current"] != 1:
                raise StatisticsImportError(
                    f"Crosswalk conflicts with existing source mapping for {mapping.source_id}"
                )

        scalar_count = 0
        structured_count = 0
        for row in validated:
            field = row["field"]
            reference = row["source_reference"] or source_url
            if field["storage"] == "scalar":
                values = (
                    row["value"], row["unit"], None, row["period"],
                    row["status"], reference, release, row["methodology"], retrieved_at, import_id,
                )
                existing = connection.execute(
                    """SELECT value, unit, observed_at, observation_period, value_status,
                              source_reference, source_version, methodology, retrieved_at, import_id
                       FROM current_statistics WHERE game_id = ? AND statistic_key = ?""",
                    (row["game_id"], field["key"]),
                ).fetchone()
                if existing is not None and tuple(existing) != values:
                    old_import = connection.execute(
                        "SELECT dataset, retrieval_date FROM dataset_imports WHERE import_id = ?",
                        (existing["import_id"],),
                    ).fetchone()
                    if old_import is None or old_import["dataset"] != source_key:
                        raise StatisticsImportError(
                            f"Existing statistic conflicts with a different source: {row['game_id']}/{field['key']}"
                        )
                    if existing["retrieved_at"]:
                        current_timestamp = _parse_aware_timestamp(
                            existing["retrieved_at"], "existing observation retrieved_at"
                        )
                        incoming_timestamp = _parse_aware_timestamp(
                            retrieved_at, "incoming manifest retrieved_at"
                        )
                        if incoming_timestamp <= current_timestamp:
                            raise StatisticsImportError(
                                f"Refusing non-newer source snapshot to replace {row['game_id']}/{field['key']}"
                            )
                    connection.execute(
                        """UPDATE current_statistics SET value=?, unit=?, observed_at=?, observation_period=?,
                                  value_status=?, source_reference=?, source_version=?, methodology=?,
                                  retrieved_at=?, import_id=?
                           WHERE game_id=? AND statistic_key=?""",
                        (*values, row["game_id"], field["key"]),
                    )
                elif existing is None:
                    connection.execute(
                        """INSERT INTO current_statistics
                           (value, unit, observed_at, observation_period, value_status,
                            source_reference, source_version, methodology, retrieved_at,
                            import_id, game_id, statistic_key)
                           VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)""",
                        (*values, row["game_id"], field["key"]),
                    )
                scalar_count += 1
            else:
                payload = _canonical_json({"value": row["value"], "unit": row["unit"]})
                values = (
                    payload, row["status"], row["period"], reference, release,
                    row["methodology"], retrieved_at, import_id,
                )
                existing = connection.execute(
                    """SELECT payload_json, value_status, observation_period, source_reference,
                              source_version, methodology, retrieved_at, import_id
                       FROM selected_game_data WHERE game_id = ? AND data_category = 'wp1_5'
                         AND data_key = ?""",
                    (row["game_id"], field["key"]),
                ).fetchone()
                if existing is not None and tuple(existing) != values:
                    old_import = connection.execute(
                        "SELECT dataset, retrieval_date FROM dataset_imports WHERE import_id = ?",
                        (existing["import_id"],),
                    ).fetchone()
                    if old_import is None or old_import["dataset"] != source_key:
                        raise StatisticsImportError(
                            f"Existing structured data conflicts with a different source: {row['game_id']}/{field['key']}"
                        )
                    if existing["retrieved_at"]:
                        current_timestamp = _parse_aware_timestamp(
                            existing["retrieved_at"], "existing observation retrieved_at"
                        )
                        incoming_timestamp = _parse_aware_timestamp(
                            retrieved_at, "incoming manifest retrieved_at"
                        )
                        if incoming_timestamp <= current_timestamp:
                            raise StatisticsImportError(
                                f"Refusing non-newer source snapshot to replace {row['game_id']}/{field['key']}"
                            )
                    connection.execute(
                        """UPDATE selected_game_data SET payload_json=?, value_status=?, observation_period=?,
                                  source_reference=?, source_version=?, methodology=?, retrieved_at=?, import_id=?
                           WHERE game_id=? AND data_category='wp1_5' AND data_key=?""",
                        (*values, row["game_id"], field["key"]),
                    )
                elif existing is None:
                    connection.execute(
                        """INSERT INTO selected_game_data
                           (payload_json, value_status, observation_period, source_reference,
                            source_version, methodology, retrieved_at, import_id, game_id, data_category, data_key)
                           VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, 'wp1_5', ?)""",
                        (*values, row["game_id"], field["key"]),
                    )
                structured_count += 1

        issues = validate_database(connection)
        if issues:
            raise StatisticsImportError("Database validation failed: " + "; ".join(issues))
        try:
            record_application_applied(
                connection, approval_id=approval_id,
                application_id=f"wp1-5:{source_key}:{release}:{source_sha[:12]}",
                details={
                    "idempotent": False, "import_id": import_id,
                    "observations": len(validated),
                    "scalar_observations": scalar_count,
                    "structured_observations": structured_count,
                },
            )
        except ApplicationApprovalError as exc:
            raise StatisticsImportError(str(exc)) from exc
        connection.commit()
        return {
            "dataset": source_key,
            "release": release,
            "import_id": import_id,
            "observations": len(validated),
            "scalar_observations": scalar_count,
            "structured_observations": structured_count,
            "crosswalk_mappings": len(crosswalk.mappings),
            "integrity": "ok",
            "foreign_keys": "ok",
            "idempotent": False,
        }
    except sqlite3.IntegrityError as exc:
        if connection.in_transaction:
            connection.rollback()
        raise StatisticsImportError(f"SQLite rejected WP1.5 import: {exc}") from exc
    except Exception:
        if connection.in_transaction:
            connection.rollback()
        raise
    finally:
        connection.close()

Provider-specific parsers must produce the normalized JSON contract documented

class StatisticsImportError(RuntimeError):
    """Raised when a normalized WP1.5 dataset cannot be safely imported."""


def _read_object(path: str | Path, label: str) -> dict[str, Any]:
    try:
        value = json.loads(Path(path).read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as exc:
        raise StatisticsImportError(f"Cannot read {label} {path}: {exc}") from exc
    if not isinstance(value, dict):
        raise StatisticsImportError(f"{label} must be a JSON object")
    return value


def _canonical_json(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False)


def _utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z")


def _parse_aware_timestamp(value: str, label: str) -> datetime:
    try:
        parsed = datetime.fromisoformat(value.replace("Z", "+00:00"))
    except ValueError as exc:
        raise StatisticsImportError(f"{label} must be an ISO-8601 timestamp") from exc
    if parsed.tzinfo is None or parsed.utcoffset() is None:
        raise StatisticsImportError(f"{label} must include a timezone")
    return parsed


def _text(value: Any, label: str) -> str:
    if not isinstance(value, str) or not value.strip():
        raise StatisticsImportError(f"{label} must be a non-empty string")
    return value


def _validate_manifest(manifest: dict[str, Any], source_key: str) -> tuple[str, str, str, str]:
    if manifest.get("manifest_schema_version") != 1 or manifest.get("work_package") != "WP1.5":
        raise StatisticsImportError("Unsupported WP1.5 source manifest")
    if manifest.get("source_key") != source_key:
        raise StatisticsImportError("Source manifest source_key does not match normalized dataset")
    release = _text(manifest.get("release"), "manifest.release")
    sha256 = _text(manifest.get("sha256"), "manifest.sha256")
    if len(sha256) != 64 or any(c not in "0123456789abcdef" for c in sha256):
        raise StatisticsImportError("manifest.sha256 must be a lowercase SHA-256 digest")
    raw_path_value = _text(manifest.get("local_raw_path"), "manifest.local_raw_path")
    raw_file = Path(raw_path_value)
    if not raw_file.is_absolute():
        raw_file = (REPOSITORY_ROOT / raw_file).resolve()
    else:
        raw_file = raw_file.resolve()
    try:
        relative_raw_path = raw_file.relative_to(REPOSITORY_ROOT.resolve())
    except ValueError as exc:
        raise StatisticsImportError("Captured raw source file must be inside the repository") from exc
    if not relative_raw_path.parts or relative_raw_path.parts[0] != "raw_data":
        raise StatisticsImportError("Captured raw source file must be directly under the repository raw_data tree")
    try:
        raw_bytes = raw_file.read_bytes()
    except OSError as exc:
        raise StatisticsImportError(f"Cannot read captured raw source file: {raw_file}") from exc
    actual_sha = hashlib.sha256(raw_bytes).hexdigest()
    if actual_sha != sha256:
        raise StatisticsImportError("Captured raw source checksum does not match its manifest")
    if not raw_bytes or manifest.get("byte_count") != len(raw_bytes):
        raise StatisticsImportError("Captured raw source byte count does not match its manifest")
    source_url = _text(manifest.get("final_url") or manifest.get("source_url"), "manifest source URL")
    if not source_url.startswith("https://"):
        raise StatisticsImportError("Source manifest URL must use HTTPS")
    retrieved_at = _text(manifest.get("retrieved_at"), "manifest.retrieved_at")
    retrieval_precision = manifest.get("retrieval_date_precision")
    if retrieval_precision == "day":
        if len(retrieved_at) != 10:
            raise StatisticsImportError(
                "day-precision manifest.retrieved_at must use YYYY-MM-DD without an inferred time"
            )
        try:
            parsed_date = date.fromisoformat(retrieved_at)
        except ValueError as exc:
            raise StatisticsImportError(
                "day-precision manifest.retrieved_at must be an ISO-8601 calendar date"
            ) from exc
        if parsed_date.isoformat() != retrieved_at:
            raise StatisticsImportError(
                "day-precision manifest.retrieved_at must use YYYY-MM-DD without an inferred time"
            )
    elif retrieval_precision in {None, "instant"}:
        try:
            retrieved_timestamp = datetime.fromisoformat(retrieved_at.replace("Z", "+00:00"))
        except ValueError as exc:
            raise StatisticsImportError("manifest.retrieved_at must be an ISO-8601 timestamp") from exc
        if retrieved_timestamp.tzinfo is None or retrieved_timestamp.utcoffset() is None:
            raise StatisticsImportError("manifest.retrieved_at must include a timezone")
    else:
        raise StatisticsImportError(
            "manifest.retrieval_date_precision must be 'instant' or 'day' when supplied"
        )
    additional_inputs = manifest.get("additional_inputs", [])
    if not isinstance(additional_inputs, list):
        raise StatisticsImportError("manifest.additional_inputs must be an array")
    for index, item in enumerate(additional_inputs):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.additional_inputs[{index}] must be an object")
        additional_path_value = _text(item.get("local_raw_path"), f"manifest.additional_inputs[{index}].local_raw_path")
        additional_path = Path(additional_path_value)
        if not additional_path.is_absolute():
            additional_path = (REPOSITORY_ROOT / additional_path).resolve()
        else:
            additional_path = additional_path.resolve()
        try:
            additional_relative = additional_path.relative_to(REPOSITORY_ROOT.resolve())
        except ValueError as exc:
            raise StatisticsImportError("Additional source snapshots must be inside the repository") from exc
        if not additional_relative.parts or additional_relative.parts[0] != "raw_data":
            raise StatisticsImportError("Additional source snapshots must be under the repository raw_data tree")
        try:
            additional_bytes = additional_path.read_bytes()
        except OSError as exc:
            raise StatisticsImportError(f"Cannot read additional source snapshot: {additional_path}") from exc
        additional_sha = _text(item.get("sha256"), f"manifest.additional_inputs[{index}].sha256")
        if len(additional_sha) != 64 or any(c not in "0123456789abcdef" for c in additional_sha):
            raise StatisticsImportError(f"manifest.additional_inputs[{index}].sha256 is invalid")
        if hashlib.sha256(additional_bytes).hexdigest() != additional_sha:
            raise StatisticsImportError(f"Additional source snapshot checksum mismatch: {additional_path}")
        if item.get("byte_count") != len(additional_bytes):
            raise StatisticsImportError(f"Additional source snapshot byte count mismatch: {additional_path}")
    upstream_manifests = manifest.get("upstream_manifests", [])
    if not isinstance(upstream_manifests, list):
        raise StatisticsImportError("manifest.upstream_manifests must be an array")
    for index, item in enumerate(upstream_manifests):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.upstream_manifests[{index}] must be an object")
        manifest_path_value = _text(item.get("path"), f"manifest.upstream_manifests[{index}].path")
        upstream_path = Path(manifest_path_value)
        if not upstream_path.is_absolute():
            upstream_path = (REPOSITORY_ROOT / upstream_path).resolve()
        else:
            upstream_path = upstream_path.resolve()
        try:
            upstream_relative = upstream_path.relative_to(REPOSITORY_ROOT.resolve())
        except ValueError as exc:
            raise StatisticsImportError("Upstream acquisition manifests must be inside the repository") from exc
        if not upstream_path.is_file():
            raise StatisticsImportError(f"Upstream acquisition manifest is missing: {upstream_path}")
        if not upstream_relative.parts:
            raise StatisticsImportError("Invalid upstream acquisition manifest path")
        try:
            upstream_digest = hashlib.sha256(upstream_path.read_bytes()).hexdigest()
        except OSError as exc:
            raise StatisticsImportError(f"Cannot read upstream acquisition manifest: {upstream_path}") from exc
        expected_digest = _text(
            item.get("sha256"), f"manifest.upstream_manifests[{index}].sha256"
        )
        if upstream_digest != expected_digest:
            raise StatisticsImportError(f"Upstream acquisition manifest checksum mismatch: {upstream_path}")
    for optional_key in ("attribution", "license_status"):
        optional_value = manifest.get(optional_key)
        if optional_value is not None and not isinstance(optional_value, str):
            raise StatisticsImportError(f"manifest.{optional_key} must be text when supplied")
    terms_url = manifest.get("terms_url")
    if terms_url is not None and (not isinstance(terms_url, str) or not terms_url.startswith("https://")):
        raise StatisticsImportError("manifest.terms_url must be HTTPS when supplied")
    return release, sha256, source_url, retrieved_at


def _validate_observation(
    row: Any,
    *,
    index: int,
    field_by_id: dict[int, dict[str, Any]],
    source_key: str,
    crosswalk_by_game: dict[str, str],
) -> dict[str, Any]:
    label = f"observations[{index}]"
    if not isinstance(row, dict):
        raise StatisticsImportError(f"{label} must be an object")
    game_id = _text(row.get("game_id"), f"{label}.game_id")
    source_id = _text(row.get("source_id"), f"{label}.source_id")
    if crosswalk_by_game.get(game_id) != source_id:
        raise StatisticsImportError(f"{label} game_id/source_id pair is absent from the explicit crosswalk")
    field_id = row.get("field_id")
    if isinstance(field_id, bool) or not isinstance(field_id, int) or field_id not in field_by_id:
        raise StatisticsImportError(f"{label}.field_id is not in the approved 41-field contract")
    field = field_by_id[field_id]
    if field["primary_source"] != source_key:
        raise StatisticsImportError(
            f"{label} source {source_key!r} is not the contracted primary source for field {field_id}"
        )
    period = _text(row.get("observation_period"), f"{label}.observation_period")
    status = row.get("value_status")
    if not isinstance(status, str) or status not in ALLOWED_VALUE_STATUSES:
        raise StatisticsImportError(f"{label}.value_status is invalid")
    if "value" not in row:
        raise StatisticsImportError(f"{label}.value must be present; use explicit JSON null only for unavailable/not_applicable")
    value = row.get("value")
    unit = row.get("unit")
    if not isinstance(unit, str) or not unit.strip():
        raise StatisticsImportError(f"{label}.unit must be a non-empty string")
    contracted_unit = field.get("unit")
    if not isinstance(contracted_unit, str):
        raise StatisticsImportError(f"Field contract unit is invalid for field {field_id}")
    allowed_units = {contracted_unit, *(part.strip() for part in contracted_unit.split("_or_") if part.strip())}
    if unit not in allowed_units:
        raise StatisticsImportError(
            f"{label}.unit {unit!r} is not permitted by field contract {contracted_unit!r}"
        )
    if field["storage"] == "scalar":
        if value is not None:
            if isinstance(value, bool) or not isinstance(value, (int, float)):
                raise StatisticsImportError(f"{label} scalar value must be a finite number or explicit NULL")
            try:
                finite = math.isfinite(float(value))
            except (OverflowError, TypeError, ValueError):
                finite = False
            if not finite:
                raise StatisticsImportError(f"{label} scalar value must be a finite number or explicit NULL")
            value = float(value)
        if value is None and status not in {"unavailable", "not_applicable"}:
            raise StatisticsImportError(f"{label} NULL value requires unavailable/not_applicable status")
        if not validate_observation_value(value, status, "scalar"):
            raise StatisticsImportError(f"{label} scalar has an invalid value/status combination")
    else:
        if value is None or not isinstance(value, (dict, list)):
            raise StatisticsImportError(f"{label} structured value must be a JSON object or array")
        if not validate_observation_value(value, status, "structured"):
            raise StatisticsImportError(f"{label} structured value requires a valid observed/estimated/projected status")
        try:
            _canonical_json(value)
        except (TypeError, ValueError) as exc:
            raise StatisticsImportError(f"{label} structured value is not valid finite JSON") from exc
    source_reference = row.get("source_reference")
    if source_reference is not None and (
        not isinstance(source_reference, str) or not source_reference.strip()
    ):
        raise StatisticsImportError(f"{label}.source_reference must be non-empty text when supplied")
    methodology = row.get("methodology", field.get("method"))
    if not isinstance(methodology, str) or not methodology.strip():
        raise StatisticsImportError(f"{label}.methodology must be non-empty text")
    return {
        "game_id": game_id,
        "source_id": source_id,
        "field": field,
        "value": value,
        "unit": unit,
        "status": status,
        "period": period,
        "source_reference": source_reference,
        "methodology": methodology,
    }


def build_wp1_5_application_artifact(
    *,
    observations_path: str | Path,
    crosswalk_path: str | Path,
    manifest_path: str | Path,
    config_path: str | Path = DEFAULT_CONFIG,
) -> dict[str, Any]:
    """Fingerprint the complete WP1.5 input set that an approver reviewed."""
    manifest = _read_object(manifest_path, "source manifest")
    normalized = _read_object(observations_path, "normalized observations")
    source_key = _text(normalized.get("source_key"), "normalized.source_key")
    dataset = _text(normalized.get("dataset", source_key), "normalized.dataset")
    if manifest.get("dataset", source_key) != dataset:
        raise StatisticsImportError("Normalized dataset identifier does not match source manifest")
    _validate_manifest(manifest, source_key)
    if normalized.get("release") != manifest.get("release"):
        raise StatisticsImportError("Normalized observations and manifest release do not match")
    if normalized.get("source_sha256") != manifest.get("sha256"):
        raise StatisticsImportError("Normalized observations and manifest checksum do not match")
    raw_path = Path(_text(manifest.get("local_raw_path"), "manifest.local_raw_path"))
    if not raw_path.is_absolute():
        raw_path = (REPOSITORY_ROOT / raw_path).resolve()
    inputs: list[Path] = [
        Path(observations_path), Path(crosswalk_path), Path(manifest_path),
        Path(config_path), raw_path,
    ]
    additional_inputs = manifest.get("additional_inputs", [])
    if not isinstance(additional_inputs, list):
        raise StatisticsImportError("manifest.additional_inputs must be an array")
    for index, item in enumerate(additional_inputs):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.additional_inputs[{index}] must be an object")
        additional_path = Path(_text(
            item.get("local_raw_path"),
            f"manifest.additional_inputs[{index}].local_raw_path",
        ))
        inputs.append(additional_path)
    upstream_manifests = manifest.get("upstream_manifests", [])
    if not isinstance(upstream_manifests, list):
        raise StatisticsImportError("manifest.upstream_manifests must be an array")
    for index, item in enumerate(upstream_manifests):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.upstream_manifests[{index}] must be an object")
        inputs.append(Path(_text(
            item.get("path"), f"manifest.upstream_manifests[{index}].path"
        )))
    selection_policy = REPOSITORY_ROOT / "config" / "source_selection.json"
    policy_version = None
    if selection_policy.is_file():
        inputs.append(selection_policy)
        policy = _read_object(selection_policy, "source-selection policy")
        policy_version = policy.get("policy_version")
    return build_application_artifact(
        domain="wp1_5_statistics",
        inputs=inputs,
        metadata={
            "source_key": normalized.get("source_key"),
            "dataset": normalized.get("dataset", normalized.get("source_key")),
            "release": normalized.get("release"),
            "source_sha256": normalized.get("source_sha256"),
            "policy_version": policy_version,
        },
    )


def validate_wp1_5_dataset(
    *,
    observations_path: str | Path,
    crosswalk_path: str | Path,
    manifest_path: str | Path,
    config_path: str | Path | None = None,
) -> dict[str, Any]:
    """Validate a normalized WP1.5 dataset without opening or changing a database."""
    selected_config_path = config_path if config_path is not None else DEFAULT_CONFIG
    normalized = _read_object(observations_path, "normalized observations")
    manifest = _read_object(manifest_path, "source manifest")
    config = _read_object(selected_config_path, "WP1.5 field contract")
    if normalized.get("schema_version") != 1 or normalized.get("work_package") != "WP1.5":
        raise StatisticsImportError("Unsupported normalized observation schema")

    source_key = _text(normalized.get("source_key"), "normalized.source_key")
    registry_reference = _text(config.get("source_registry"), "WP1.5 contract source_registry")
    registry_path = Path(registry_reference)
    if not registry_path.is_absolute():
        registry_path = (REPOSITORY_ROOT / registry_path).resolve()
    try:
        source_registry = load_registry(registry_path)
    except SourceError as exc:
        raise StatisticsImportError(f"Cannot load project source registry: {exc}") from exc
    canonical_source_key = canonical_source_id(source_registry, source_key)
    if canonical_source_key is None:
        raise StatisticsImportError(f"Unknown source ID in project source registry: {source_key}")

    dataset = _text(normalized.get("dataset", source_key), "normalized.dataset")
    if manifest.get("dataset", source_key) != dataset:
        raise StatisticsImportError("Normalized dataset identifier does not match source manifest")
    release, source_sha, source_url, retrieved_at = _validate_manifest(manifest, source_key)
    if normalized.get("source_key") != manifest.get("source_key"):
        raise StatisticsImportError("Normalized observations and source manifest do not match")
    if normalized.get("source_sha256") != source_sha or normalized.get("release") != release:
        raise StatisticsImportError("Normalized observations do not identify the captured source release/checksum")

    fields = config.get("fields")
    if not isinstance(fields, list) or len(fields) != 41:
        raise StatisticsImportError("WP1.5 contract must enumerate exactly 41 fields")
    field_by_id = {item.get("id"): item for item in fields if isinstance(item, dict)}
    if len(field_by_id) != 41:
        raise StatisticsImportError("WP1.5 contract must enumerate exactly 41 distinct fields")

    try:
        crosswalk = load_crosswalk(crosswalk_path, expected_dataset=dataset)
    except CrosswalkError as exc:
        raise StatisticsImportError(f"Invalid source crosswalk: {exc}") from exc
    if crosswalk.dataset != dataset:
        raise StatisticsImportError("Crosswalk dataset does not match normalized dataset")
    if crosswalk.source_key != source_key:
        raise StatisticsImportError("Crosswalk source_key does not match source contract")
    crosswalk_by_game = {item.game_id: item.source_id for item in crosswalk.mappings}

    rows = normalized.get("observations")
    if not isinstance(rows, list) or not rows:
        raise StatisticsImportError("Normalized observations must be a non-empty array")
    validated = [
        _validate_observation(
            row,
            index=index,
            field_by_id=field_by_id,
            source_key=canonical_source_key,
            crosswalk_by_game=crosswalk_by_game,
        )
        for index, row in enumerate(rows)
    ]
    keys = [(row["game_id"], row["field"]["id"]) for row in validated]
    if len(keys) != len(set(keys)):
        raise StatisticsImportError("Duplicate country/field observations in normalized dataset")

    try:
        artifact = build_wp1_5_application_artifact(
            observations_path=observations_path,
            crosswalk_path=crosswalk_path,
            manifest_path=manifest_path,
            config_path=selected_config_path,
        )
    except (ApplicationApprovalError, OSError, ValueError) as exc:
        raise StatisticsImportError(f"Cannot fingerprint WP1.5 application inputs: {exc}") from exc

    # Re-read all validated JSON inputs to detect changes during validation.
    if _read_object(observations_path, "normalized observations") != normalized:
        raise StatisticsImportError("Normalized observations changed during dry-run validation")
    if _read_object(manifest_path, "source manifest") != manifest:
        raise StatisticsImportError("Source manifest changed during dry-run validation")
    if _read_object(selected_config_path, "WP1.5 field contract") != config:
        raise StatisticsImportError("WP1.5 field contract changed during dry-run validation")
    _validate_manifest(manifest, source_key)

    field_counts: dict[str, int] = {}
    for row in validated:
        field_id = str(row["field"]["id"])
        field_counts[field_id] = field_counts.get(field_id, 0) + 1
    return {
        "assessment": "wp1_5_statistics_import_dry_run",
        "source_key": source_key,
        "canonical_source_key": canonical_source_key,
        "dataset": dataset,
        "release": release,
        "source_sha256": source_sha,
        "source_url": source_url,
        "retrieved_at": retrieved_at,
        "observations": len(validated),
        "crosswalk_mappings": len(crosswalk.mappings),
        "field_counts": dict(sorted(field_counts.items(), key=lambda item: int(item[0]))),
        "license_status": manifest.get("license_status"),
        "attribution": manifest.get("attribution"),
        "application_artifact_sha256": artifact["artifact_sha256"],
        "application_artifact": artifact,
        "side_effects": {
            "database_opened": False,
            "database_written": False,
            "approval_created": False,
            "source_selection_changed": False,
        },
    }


def import_wp1_5_dataset(
    *,
    observations_path: str | Path,
    crosswalk_path: str | Path,
    manifest_path: str | Path,
    database_path: str | Path = DEFAULT_DATABASE_PATH,

    approval_id: str,

    normalized = _read_object(observations_path, "normalized observations")
    manifest = _read_object(manifest_path, "source manifest")
    config = _read_object(config_path, "WP1.5 field contract")
    if normalized.get("schema_version") != 1 or normalized.get("work_package") != "WP1.5":
        raise StatisticsImportError("Unsupported normalized observation schema")
    source_key = _text(normalized.get("source_key"), "normalized.source_key")
    registry_reference = _text(config.get("source_registry"), "WP1.5 contract source_registry")
    registry_path = Path(registry_reference)
    if not registry_path.is_absolute():
        # Registry references in project configuration are repository-root-relative,
        # not relative to the directory containing the referencing config file.
        registry_path = (REPOSITORY_ROOT / registry_path).resolve()
    try:
        source_registry = load_registry(registry_path)
    except SourceError as exc:
        raise StatisticsImportError(f"Cannot load project source registry: {exc}") from exc
    canonical_source_key = canonical_source_id(source_registry, source_key)
    if canonical_source_key is None:
        raise StatisticsImportError(f"Unknown source ID in project source registry: {source_key}")
    # Keep the historical ID in manifests, crosswalks and database provenance;
    # use its canonical provider ID only for current field-contract validation.
    dataset = _text(normalized.get("dataset", source_key), "normalized.dataset")
    if manifest.get("dataset", source_key) != dataset:
        raise StatisticsImportError("Normalized dataset identifier does not match source manifest")
    release, source_sha, source_url, retrieved_at = _validate_manifest(manifest, source_key)
    if normalized.get("source_key") != manifest.get("source_key"):
        raise StatisticsImportError("Normalized observations and source manifest do not match")
    if normalized.get("source_sha256") != source_sha or normalized.get("release") != release:
        raise StatisticsImportError("Normalized observations do not identify the captured source release/checksum")

    fields = config.get("fields")
    if not isinstance(fields, list):
        raise StatisticsImportError("WP1.5 contract fields must be an array")
    field_by_id = {item.get("id"): item for item in fields if isinstance(item, dict)}

        raise StatisticsImportError("Crosswalk dataset does not match normalized dataset")

            source_key=canonical_source_key,
            crosswalk_by_game=crosswalk_by_game,

"""Normalize the immutable WDI WP1.5 area/density snapshot for review/import.

This adapter emits candidate observations only. It does not approve, select, or
write data to either project database.
"""
from __future__ import annotations

import argparse

from tools.acquire_world_bank_wdi_wp1_5_area import (
    API, CROSSWALK_DATASET, DATASET_KEY, INDICATORS, ROOT, SOURCE_KEY, AreaSnapshotError, load_snapshot,
)

def _value_status(provider_status: Any) -> str:
    """Keep blank WDI statuses unclassified; require review for coded statuses."""
    if provider_status is None or provider_status == "":
        return "unclassified"
    raise AreaSnapshotError(
        f"WDI obs_status {provider_status!r} requires an explicit canonical status mapping"
    )

def _write_json(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("x", encoding="utf-8") as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write("\n")

def normalize_snapshot(
    manifest_path: str | Path,
    crosswalk_path: str | Path,
    observations_path: str | Path,
    import_manifest_path: str | Path,
    *,
    repository_root: str | Path = ROOT,
) -> dict[str, Any]:
    root = Path(repository_root).resolve()
    loaded = load_snapshot(manifest_path, repository_root=root)
    crosswalk = load_crosswalk(
        Path(crosswalk_path) if Path(crosswalk_path).is_absolute() else root / crosswalk_path,
        expected_dataset=CROSSWALK_DATASET,
    )
    if crosswalk.source_key != SOURCE_KEY:
        raise AreaSnapshotError("Crosswalk source_key does not match the WDI provider")
    mapping = {item.source_id: item.game_id for item in crosswalk.mappings}
    if len(mapping) != 187:
        raise AreaSnapshotError(f"Expected the owner-approved 187-entry WDI crosswalk; found {len(mapping)}")
    if any(not (len(code) == 3 and code.isupper() and code.isalpha()) for code in mapping):
        raise AreaSnapshotError("Crosswalk source IDs must be uppercase three-letter WDI codes")

    rows: list[dict[str, Any]] = []
    per_indicator: dict[str, int] = {}
    for code, spec in INDICATORS.items():
        latest: dict[str, dict[str, Any]] = {}
        for item in loaded["observations"][code]:
            iso3 = item.get("countryiso3code")
            if not isinstance(iso3, str) or len(iso3) != 3:
                continue
            country = loaded["countries"].get(iso3)
            if country is None or (country.get("region") or {}).get("id") == "NA":
                continue
            value = item.get("value")
            if value is None or iso3 not in mapping:
                continue
            try:
                year = int(item.get("date"))
                numeric = float(value)
            except (TypeError, ValueError) as exc:
                raise AreaSnapshotError(f"Invalid selected value for {code}/{iso3}") from exc
            if year < 1900 or not validate_observation_value(numeric, "unclassified", "scalar"):
                raise AreaSnapshotError(f"Invalid selected value for {code}/{iso3}/{year}")
            previous = latest.get(iso3)
            if previous is None or year > previous["year"]:
                latest[iso3] = {
                    "year": year, "value": numeric,
                    "provider_status": item.get("obs_status"),
                    "country_name": (item.get("country") or {}).get("value"),
                }
        if set(latest) != set(mapping):
            missing = sorted(set(mapping) - set(latest))
            raise AreaSnapshotError(f"{code}: mapped entities lack non-null observations: {missing[:10]}")
        per_indicator[code] = len(latest)
        metadata = loaded["indicator_metadata"][code]
        for iso3, selected in sorted(latest.items()):
            rows.append({
                "game_id": mapping[iso3],
                "source_id": iso3,
                "field_id": spec["field_id"],
                "value": selected["value"],
                "unit": spec["unit"],
                "value_status": _value_status(selected["provider_status"]),
                "observation_period": str(selected["year"]),
                "source_reference": f"{API}/country/{iso3}/indicator/{code}?date={selected['year']}",
                "methodology": (
                    f"World Bank WDI indicator {code}: {spec['meaning']} "
                    f"Originating source organization: {metadata.get('sourceOrganization') or 'not supplied'}. "
                    "Provider obs_status is blank; the populated scalar value remains unclassified."
                ),
                "provenance": {
                    "wdi_iso3": iso3,
                    "country_name": selected["country_name"],
                    "indicator_code": code,
                    "provider_obs_status": selected["provider_status"],
                    "provider_source_organization": metadata.get("sourceOrganization"),
                    "provider_source_note": metadata.get("sourceNote"),
                    "provider_unit_metadata": metadata.get("unit"),

"""Canonical value/status rules shared by acquisition and selection pipelines.

Missing observations are represented by absent rows. Explicit null values are
reserved for unavailable or not-applicable observations.
"""
from __future__ import annotations

import json
import math
from typing import Any

UNCLASSIFIED = "unclassified"
NULL_VALUE_STATUSES = frozenset({"unavailable", "not_applicable"})
CLASSIFIED_POPULATED_STATUSES = frozenset({"observed", "estimated", "projected"})
POPULATED_SCALAR_STATUSES = CLASSIFIED_POPULATED_STATUSES | {UNCLASSIFIED}
ALLOWED_VALUE_STATUSES = POPULATED_SCALAR_STATUSES | NULL_VALUE_STATUSES
POPULATED_STRUCTURED_STATUSES = CLASSIFIED_POPULATED_STATUSES


def validate_observation_value(value: Any, status: Any, storage: Any) -> bool:
    """Validate a scalar or structured value/status pair without inference.

    ``unclassified`` is valid only for finite populated scalar values. Structured
    payloads require a classified status. Explicit nulls require unavailable or
    not_applicable; missing observations are represented by absent rows.
    """
    if storage not in {"scalar", "structured"} or not isinstance(status, str) or status not in ALLOWED_VALUE_STATUSES:
        return False
    if value is None:
        return status in NULL_VALUE_STATUSES
    if storage == "scalar":
        if status not in POPULATED_SCALAR_STATUSES:
            return False
        if isinstance(value, bool) or not isinstance(value, (int, float)):
            return False
        try:
            return math.isfinite(float(value))
        except (OverflowError, TypeError, ValueError):
            return False
    if status not in POPULATED_STRUCTURED_STATUSES or not isinstance(value, (dict, list)):
        return False
    try:
        json.dumps(value, ensure_ascii=False, allow_nan=False)
    except (TypeError, ValueError):
        return False
    return True

"""Transactional, local-only SQLite schema migration management."""

from __future__ import annotations

import re
import sqlite3
from datetime import datetime, timezone
from pathlib import Path

from tools.database.connection import REPOSITORY_ROOT

DEFAULT_MIGRATIONS_DIR = REPOSITORY_ROOT / "migrations"
MIGRATION_PATTERN = re.compile(r"^(\d{3})_([a-z0-9_]+)\.sql$")


class MigrationError(RuntimeError):
    """Raised when the local database cannot be safely migrated."""


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z")


def _migration_files(migrations_dir: str | Path) -> list[tuple[int, str, Path]]:
    directory = Path(migrations_dir)
    if not directory.is_dir():
        raise MigrationError(f"Migration directory does not exist: {directory}")
    migrations = []
    for path in directory.iterdir():
        match = MIGRATION_PATTERN.fullmatch(path.name)
        if match and path.is_file():
            migrations.append((int(match.group(1)), path.name, path))
    migrations.sort(key=lambda item: item[0])
    versions = [item[0] for item in migrations]
    if len(versions) != len(set(versions)):
        raise MigrationError("Migration filenames contain duplicate version numbers")
    if versions and versions != list(range(1, max(versions) + 1)):
        raise MigrationError("Migration versions must be contiguous starting at 001")
    return migrations


def latest_schema_version(migrations_dir: str | Path = DEFAULT_MIGRATIONS_DIR) -> int:
    """Return the highest registered migration version (zero when none exist)."""
    migrations = _migration_files(migrations_dir)
    return migrations[-1][0] if migrations else 0


def current_schema_version(connection: sqlite3.Connection) -> int:
    exists = connection.execute(
        "SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'schema_migrations'"
    ).fetchone()
    if exists is None:
        return 0
    row = connection.execute("SELECT COALESCE(MAX(version), 0) AS version FROM schema_migrations").fetchone()
    return int(row["version"])


def apply_migrations(
    connection: sqlite3.Connection,
    migrations_dir: str | Path = DEFAULT_MIGRATIONS_DIR,
) -> int:
    """Apply pending migrations in order, one SQLite transaction per migration."""
    migrations = _migration_files(migrations_dir)
    connection.execute(
        """CREATE TABLE IF NOT EXISTS schema_migrations (
            version INTEGER PRIMARY KEY CHECK (version > 0),
            name TEXT NOT NULL UNIQUE,
            applied_at TEXT NOT NULL
        )"""
    )
    applied_version = current_schema_version(connection)
    latest_available = migrations[-1][0] if migrations else 0
    if applied_version > latest_available:
        raise MigrationError(
            f"Database schema version {applied_version} is newer than supported version {latest_available}"
        )

    applied_rows = connection.execute(
        "SELECT version, name FROM schema_migrations ORDER BY version"
    ).fetchall()
    applied_versions = [int(row["version"]) for row in applied_rows]
    if applied_versions != list(range(1, applied_version + 1)):
        raise MigrationError("Applied migration ledger is not contiguous starting at version 1")
    for row in applied_rows:
        matching = next((item for item in migrations if item[0] == row["version"]), None)
        if matching is None or matching[1] != row["name"]:
            raise MigrationError(
                f"Applied migration {row['version']} ({row['name']}) does not match available migration files"
            )

    for version, name, path in migrations:
        if version <= applied_version:
            continue
        sql = path.read_text(encoding="utf-8")
        try:
            connection.executescript(
                "BEGIN IMMEDIATE;\n"
                + sql
                + "\nINSERT INTO schema_migrations(version, name, applied_at) VALUES ("
                + str(version)
                + ", "
                + "'"
                + name.replace("'", "''")
                + "', '"
                + utc_now()
                + "');\nCOMMIT;"
            )
        except sqlite3.Error as exc:
            if connection.in_transaction:
                connection.rollback()
            raise MigrationError(f"Migration {name} failed: {exc}") from exc
        applied_version = version
    return applied_version


def validate_database(connection: sqlite3.Connection) -> list[str]:
    """Return SQLite integrity and foreign-key violations, if any."""
    issues: list[str] = []
    integrity = connection.execute("PRAGMA integrity_check").fetchall()
    issues.extend(str(row[0]) for row in integrity if row[0] != "ok")
    foreign_keys = connection.execute("PRAGMA foreign_key_check").fetchall()
    issues.extend(
        f"foreign key violation: table={row[0]} rowid={row[1]} parent={row[2]}"
        for row in foreign_keys
    )
    return issues
