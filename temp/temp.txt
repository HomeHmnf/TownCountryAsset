import sqlite3
from tools.database.connection import DEFAULT_DATABASE_PATH

con = sqlite3.connect(DEFAULT_DATABASE_PATH)
con.row_factory = sqlite3.Row

sql = """
SELECT ne.game_id, ne.source_id AS wdi_iso3, wpp.source_id AS wpp_m49,
       c.entity_type, c.iso_assignment_status
FROM country_sources ne
JOIN country_sources wpp ON wpp.game_id = ne.game_id
JOIN countries c ON c.game_id = ne.game_id
WHERE ne.dataset = ? AND wpp.dataset = ?
  AND ne.is_current = 1 AND wpp.is_current = 1
  AND length(ne.source_id) = 3
  AND ne.source_id NOT LIKE "%-%"
ORDER BY ne.source_id
"""

rows = con.execute(sql, ("natural_earth_admin_0", "un_wpp")).fetchall()
print("MAPPING COUNT:", len(rows))
print("DISTINCT WDI ISO3:", len({r["wdi_iso3"] for r in rows}))
print("DISTINCT GAME IDs:", len({r["game_id"] for r in rows}))
print("\nMAPPINGS:")
for row in rows:
    print(dict(row))

con.close()
