class StatisticsImportError(RuntimeError):
    """Raised when a normalized WP1.5 dataset cannot be safely imported."""


def _read_object(path: str | Path, label: str) -> dict[str, Any]:
    try:
        value = json.loads(Path(path).read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as exc:
        raise StatisticsImportError(f"Cannot read {label} {path}: {exc}") from exc
    if not isinstance(value, dict):
        raise StatisticsImportError(f"{label} must be a JSON object")
    return value


def _canonical_json(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False)


def _utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z")


def _parse_aware_timestamp(value: str, label: str) -> datetime:
    try:
        parsed = datetime.fromisoformat(value.replace("Z", "+00:00"))
    except ValueError as exc:
        raise StatisticsImportError(f"{label} must be an ISO-8601 timestamp") from exc
    if parsed.tzinfo is None or parsed.utcoffset() is None:
        raise StatisticsImportError(f"{label} must include a timezone")
    return parsed


def _text(value: Any, label: str) -> str:
    if not isinstance(value, str) or not value.strip():
        raise StatisticsImportError(f"{label} must be a non-empty string")
    return value


def _validate_manifest(manifest: dict[str, Any], source_key: str) -> tuple[str, str, str, str]:
    if manifest.get("manifest_schema_version") != 1 or manifest.get("work_package") != "WP1.5":
        raise StatisticsImportError("Unsupported WP1.5 source manifest")
    if manifest.get("source_key") != source_key:
        raise StatisticsImportError("Source manifest source_key does not match normalized dataset")
    release = _text(manifest.get("release"), "manifest.release")
    sha256 = _text(manifest.get("sha256"), "manifest.sha256")
    if len(sha256) != 64 or any(c not in "0123456789abcdef" for c in sha256):
        raise StatisticsImportError("manifest.sha256 must be a lowercase SHA-256 digest")
    raw_path_value = _text(manifest.get("local_raw_path"), "manifest.local_raw_path")
    raw_file = Path(raw_path_value)
    if not raw_file.is_absolute():
        raw_file = (REPOSITORY_ROOT / raw_file).resolve()
    else:
        raw_file = raw_file.resolve()
    try:
        relative_raw_path = raw_file.relative_to(REPOSITORY_ROOT.resolve())
    except ValueError as exc:
        raise StatisticsImportError("Captured raw source file must be inside the repository") from exc
    if not relative_raw_path.parts or relative_raw_path.parts[0] != "raw_data":
        raise StatisticsImportError("Captured raw source file must be directly under the repository raw_data tree")
    try:
        raw_bytes = raw_file.read_bytes()
    except OSError as exc:
        raise StatisticsImportError(f"Cannot read captured raw source file: {raw_file}") from exc
    actual_sha = hashlib.sha256(raw_bytes).hexdigest()
    if actual_sha != sha256:
        raise StatisticsImportError("Captured raw source checksum does not match its manifest")
    if not raw_bytes or manifest.get("byte_count") != len(raw_bytes):
        raise StatisticsImportError("Captured raw source byte count does not match its manifest")
    source_url = _text(manifest.get("final_url") or manifest.get("source_url"), "manifest source URL")
    if not source_url.startswith("https://"):
        raise StatisticsImportError("Source manifest URL must use HTTPS")
    retrieved_at = _text(manifest.get("retrieved_at"), "manifest.retrieved_at")
    try:
        retrieved_timestamp = datetime.fromisoformat(retrieved_at.replace("Z", "+00:00"))
    except ValueError as exc:
        raise StatisticsImportError("manifest.retrieved_at must be an ISO-8601 timestamp") from exc
    if retrieved_timestamp.tzinfo is None or retrieved_timestamp.utcoffset() is None:
        raise StatisticsImportError("manifest.retrieved_at must include a timezone")
    additional_inputs = manifest.get("additional_inputs", [])
    if not isinstance(additional_inputs, list):
        raise StatisticsImportError("manifest.additional_inputs must be an array")
    for index, item in enumerate(additional_inputs):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.additional_inputs[{index}] must be an object")
        additional_path_value = _text(item.get("local_raw_path"), f"manifest.additional_inputs[{index}].local_raw_path")
        additional_path = Path(additional_path_value)
        if not additional_path.is_absolute():
            additional_path = (REPOSITORY_ROOT / additional_path).resolve()
        else:
            additional_path = additional_path.resolve()
        try:
            additional_relative = additional_path.relative_to(REPOSITORY_ROOT.resolve())
        except ValueError as exc:
            raise StatisticsImportError("Additional source snapshots must be inside the repository") from exc
        if not additional_relative.parts or additional_relative.parts[0] != "raw_data":
            raise StatisticsImportError("Additional source snapshots must be under the repository raw_data tree")
        try:
            additional_bytes = additional_path.read_bytes()
        except OSError as exc:
            raise StatisticsImportError(f"Cannot read additional source snapshot: {additional_path}") from exc
        additional_sha = _text(item.get("sha256"), f"manifest.additional_inputs[{index}].sha256")
        if len(additional_sha) != 64 or any(c not in "0123456789abcdef" for c in additional_sha):
            raise StatisticsImportError(f"manifest.additional_inputs[{index}].sha256 is invalid")
        if hashlib.sha256(additional_bytes).hexdigest() != additional_sha:
            raise StatisticsImportError(f"Additional source snapshot checksum mismatch: {additional_path}")
        if item.get("byte_count") != len(additional_bytes):
            raise StatisticsImportError(f"Additional source snapshot byte count mismatch: {additional_path}")
    upstream_manifests = manifest.get("upstream_manifests", [])
    if not isinstance(upstream_manifests, list):
        raise StatisticsImportError("manifest.upstream_manifests must be an array")
    for index, item in enumerate(upstream_manifests):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.upstream_manifests[{index}] must be an object")
        manifest_path_value = _text(item.get("path"), f"manifest.upstream_manifests[{index}].path")
        upstream_path = Path(manifest_path_value)
        if not upstream_path.is_absolute():
            upstream_path = (REPOSITORY_ROOT / upstream_path).resolve()
        else:
            upstream_path = upstream_path.resolve()
        try:
            upstream_relative = upstream_path.relative_to(REPOSITORY_ROOT.resolve())
        except ValueError as exc:
            raise StatisticsImportError("Upstream acquisition manifests must be inside the repository") from exc
        if not upstream_path.is_file():
            raise StatisticsImportError(f"Upstream acquisition manifest is missing: {upstream_path}")
        if not upstream_relative.parts:
            raise StatisticsImportError("Invalid upstream acquisition manifest path")
        try:
            upstream_digest = hashlib.sha256(upstream_path.read_bytes()).hexdigest()
        except OSError as exc:
            raise StatisticsImportError(f"Cannot read upstream acquisition manifest: {upstream_path}") from exc
        expected_digest = _text(
            item.get("sha256"), f"manifest.upstream_manifests[{index}].sha256"
        )
        if upstream_digest != expected_digest:
            raise StatisticsImportError(f"Upstream acquisition manifest checksum mismatch: {upstream_path}")
    for optional_key in ("attribution", "license_status"):
        optional_value = manifest.get(optional_key)
        if optional_value is not None and not isinstance(optional_value, str):
            raise StatisticsImportError(f"manifest.{optional_key} must be text when supplied")
    terms_url = manifest.get("terms_url")
    if terms_url is not None and (not isinstance(terms_url, str) or not terms_url.startswith("https://")):
        raise StatisticsImportError("manifest.terms_url must be HTTPS when supplied")
    return release, sha256, source_url, retrieved_at


def _validate_observation(
    row: Any,
    *,
    index: int,
    field_by_id: dict[int, dict[str, Any]],
    source_key: str,
    crosswalk_by_game: dict[str, str],
) -> dict[str, Any]:
    label = f"observations[{index}]"
    if not isinstance(row, dict):
        raise StatisticsImportError(f"{label} must be an object")
    game_id = _text(row.get("game_id"), f"{label}.game_id")
    source_id = _text(row.get("source_id"), f"{label}.source_id")
    if crosswalk_by_game.get(game_id) != source_id:
        raise StatisticsImportError(f"{label} game_id/source_id pair is absent from the explicit crosswalk")
    field_id = row.get("field_id")
    if isinstance(field_id, bool) or not isinstance(field_id, int) or field_id not in field_by_id:
        raise StatisticsImportError(f"{label}.field_id is not in the approved 41-field contract")
    field = field_by_id[field_id]
    if field["primary_source"] != source_key:
        raise StatisticsImportError(
            f"{label} source {source_key!r} is not the contracted primary source for field {field_id}"
        )
    period = _text(row.get("observation_period"), f"{label}.observation_period")
    status = row.get("value_status")
    if not isinstance(status, str) or status not in {"observed", "estimated", "projected", "unavailable", "not_applicable"}:
        raise StatisticsImportError(f"{label}.value_status is invalid")
    if "value" not in row:
        raise StatisticsImportError(f"{label}.value must be present; use explicit JSON null only for unavailable/not_applicable")
    value = row.get("value")
    unit = row.get("unit")
    if not isinstance(unit, str) or not unit.strip():
        raise StatisticsImportError(f"{label}.unit must be a non-empty string")
    contracted_unit = field.get("unit")
    if not isinstance(contracted_unit, str):
        raise StatisticsImportError(f"Field contract unit is invalid for field {field_id}")
    allowed_units = {contracted_unit, *(part.strip() for part in contracted_unit.split("_or_") if part.strip())}
    if unit not in allowed_units:
        raise StatisticsImportError(
            f"{label}.unit {unit!r} is not permitted by field contract {contracted_unit!r}"
        )
    if field["storage"] == "scalar":
        if value is None:
            if status not in {"unavailable", "not_applicable"}:
                raise StatisticsImportError(f"{label} NULL value requires unavailable/not_applicable status")
        else:
            if isinstance(value, bool) or not isinstance(value, (int, float)):
                raise StatisticsImportError(f"{label} scalar value must be a finite number or explicit NULL")
            try:
                finite = math.isfinite(float(value))
            except (OverflowError, ValueError):
                finite = False
            if not finite:
                raise StatisticsImportError(f"{label} scalar value must be a finite number or explicit NULL")
            value = float(value)
            if status not in {"observed", "estimated", "projected"}:
                raise StatisticsImportError(f"{label} populated scalar requires observed/estimated/projected status")
    else:
        if value is None or not isinstance(value, (dict, list)):
            raise StatisticsImportError(f"{label} structured value must be a JSON object or array")
        if status not in {"observed", "estimated", "projected"}:
            raise StatisticsImportError(f"{label} structured value requires observed/estimated/projected status")
        try:
            _canonical_json(value)
        except (TypeError, ValueError) as exc:
            raise StatisticsImportError(f"{label} structured value is not valid finite JSON") from exc
    source_reference = row.get("source_reference")
    if source_reference is not None and (
        not isinstance(source_reference, str) or not source_reference.strip()
    ):
        raise StatisticsImportError(f"{label}.source_reference must be non-empty text when supplied")
    methodology = row.get("methodology", field.get("method"))
    if not isinstance(methodology, str) or not methodology.strip():
        raise StatisticsImportError(f"{label}.methodology must be non-empty text")
    return {
        "game_id": game_id,
        "source_id": source_id,
        "field": field,
        "value": value,
        "unit": unit,
        "status": status,
        "period": period,
        "source_reference": source_reference,
        "methodology": methodology,
    }


def build_wp1_5_application_artifact(
    *,
    observations_path: str | Path,
    crosswalk_path: str | Path,
    manifest_path: str | Path,
    config_path: str | Path = DEFAULT_CONFIG,
) -> dict[str, Any]:
    """Fingerprint the complete WP1.5 input set that an approver reviewed."""
    manifest = _read_object(manifest_path, "source manifest")
    normalized = _read_object(observations_path, "normalized observations")
    source_key = _text(normalized.get("source_key"), "normalized.source_key")
    _validate_manifest(manifest, source_key)
    if normalized.get("release") != manifest.get("release"):
        raise StatisticsImportError("Normalized observations and manifest release do not match")
    if normalized.get("source_sha256") != manifest.get("sha256"):
        raise StatisticsImportError("Normalized observations and manifest checksum do not match")
    raw_path = Path(_text(manifest.get("local_raw_path"), "manifest.local_raw_path"))
    if not raw_path.is_absolute():
        raw_path = (REPOSITORY_ROOT / raw_path).resolve()
    inputs: list[Path] = [
        Path(observations_path), Path(crosswalk_path), Path(manifest_path),
        Path(config_path), raw_path,
    ]
    additional_inputs = manifest.get("additional_inputs", [])
    if not isinstance(additional_inputs, list):
        raise StatisticsImportError("manifest.additional_inputs must be an array")
    for index, item in enumerate(additional_inputs):
        if not isinstance(item, dict):
            raise StatisticsImportError(f"manifest.additional_inputs[{index}] must be an object")
        additional_path = Path(_text(
            item.get("local_raw_path"),
            f"manifest.additional_inputs[{index}].local_raw_path",
        ))
        inputs.append(additional_path)
    upstream_manifests = manifest.get("upstream_manifests", [])
    if not isinstance(upstream_manifests, list):
        raise StatisticsImportError("manifest.upstream_manifests must be an array")

class AreaSnapshotError(RuntimeError):
    """Invalid WDI WP1.5 area snapshot or candidate validation failure."""


def _url(path: str, params: dict[str, str]) -> str:
    return f"{API}/{path}?{urllib.parse.urlencode(params)}"


def _save(root: Path, raw_dir: Path, filename: str, url: str, params: dict[str, str],
          request_get: Callable[[str], tuple[bytes, int, dict[str, str], str]],
          page: int | None = None) -> tuple[dict[str, Any], Any]:
    body, status, headers, final_url = request_get(url)
    if status != 200 or not body:
        raise AreaSnapshotError(f"Invalid HTTP response ({status}) or empty body: {url}")
    path = raw_dir / filename
    path.parent.mkdir(parents=True, exist_ok=True)
    try:
        with path.open("xb") as stream:
            stream.write(body)
    except FileExistsError as exc:
        raise AreaSnapshotError(f"Refusing to overwrite {path}") from exc
    payload = common._payload(body, filename)
    artifact = {
        "filename": filename, "local_raw_path": path.relative_to(root).as_posix(),
        "content_type": next((v for k, v in headers.items() if k.casefold() == "content-type"), "application/json"),
        "format": "json", "byte_count": len(body),
        "sha256": hashlib.sha256(body).hexdigest(), "url": url, "final_url": final_url,
        "http_status": status, "retrieved_at": datetime.now(timezone.utc).isoformat(),
        "request_parameters": params, "response_headers": headers,
        "request_headers": {"User-Agent": "TownCountryAsset-WDI-WP1.5-Area/1.0", "Accept": "application/json"},
        "page_number": page,
    }
    return artifact, payload


def capture_snapshot(*, repository_root: str | Path = ROOT, start_year: int = 2015,
                     end_year: int = 2025, retrieved_at: str | None = None,
                     request_get: Callable[[str], tuple[bytes, int, dict[str, str], str]] = common._request) -> dict[str, Any]:
    if start_year < 1900 or end_year < start_year or end_year > 2100:
        raise AreaSnapshotError("Invalid snapshot year range")
    root = Path(repository_root).resolve()
    started = common._aware(retrieved_at)
    snapshot_id = "api_snapshot_" + started.strftime("%Y%m%dT%H%M%S%fZ")
    raw_dir = root / RAW_BASE / snapshot_id
    manifest_path = root / MANIFEST_BASE / f"{snapshot_id}.json"
    if raw_dir.exists() or manifest_path.exists():
        raise AreaSnapshotError(f"Snapshot already exists: {snapshot_id}")
    raw_dir.mkdir(parents=True, exist_ok=False)
    artifacts: list[dict[str, Any]] = []

    params = {"format": "json"}
    artifact, source_payload = _save(root, raw_dir, "source_metadata.json",
        _url("sources/2", params), params, request_get)
    if not any(str(row.get("id")) == SOURCE_ID and row.get("name") == "World Development Indicators"
               for row in source_payload[1]):
        raise AreaSnapshotError("API source ID 2 is not World Development Indicators")
    artifacts.append(artifact)

    country_rows: list[dict[str, Any]] = []
    page = 1
    country_pages = None
    while True:
        params = {"format": "json", "per_page": "400", "page": str(page)}
        artifact, payload = _save(root, raw_dir, f"country_metadata_page_{page:03d}.json",
            _url("country", params), params, request_get, page)
        pages = common._pages(payload, page, "country metadata")
        if country_pages is not None and pages != country_pages:
            raise AreaSnapshotError("Country metadata pagination changed during capture")
        country_pages = pages
        artifact.update({"page_count": pages, "row_count": len(payload[1])})
        artifacts.append(artifact)
        country_rows.extend(payload[1])
        if page == pages:
            break
        page += 1
    if not country_rows:
        raise AreaSnapshotError("Country metadata catalogue is empty")

    summaries = {}
    for code in INDICATORS:
        params = {"format": "json"}
        artifact, payload = _save(root, raw_dir, f"indicator_metadata_{code}.json",
            _url(f"indicator/{urllib.parse.quote(code, safe='.')}", params), params, request_get)
        meta = next((row for row in payload[1] if row.get("id") == code), None)
        if meta is None or str((meta.get("source") or {}).get("id")) != SOURCE_ID:
            raise AreaSnapshotError(f"{code} is not confirmed as a WDI indicator")
        artifacts.append(artifact)
        rows = []
        page = 1
        page_count = None
        api_last_updated = None
        while True:
            params = {"format": "json", "date": f"{start_year}:{end_year}",
                      "per_page": str(PER_PAGE), "page": str(page)}
            artifact, response = _save(root, raw_dir, f"observations_{code}_page_{page:03d}.json",
                _url(f"country/all/indicator/{urllib.parse.quote(code, safe='.')}", params),
                params, request_get, page)
            pages = common._pages(response, page, f"{code} observations")
            if page_count is not None and pages != page_count:
                raise AreaSnapshotError(f"{code}: pagination changed during capture")
            page_count = pages
            api_last_updated = response[0].get("lastupdated") or api_last_updated
            if any((row.get("indicator") or {}).get("id") not in (None, code) for row in response[1]):
                raise AreaSnapshotError(f"{code}: unexpected indicator in observation response")
            artifact.update({"page_count": pages, "row_count": len(response[1])})
            artifacts.append(artifact)
            rows.extend(response[1])
            if page == pages:
                break
            page += 1
        summaries[code] = {
            "field_id": INDICATORS[code]["field_id"], "pages": page_count,
            "observation_rows": len(rows), "api_last_updated": api_last_updated,
            "indicator_name": meta.get("name"), "unit_metadata": meta.get("unit"),
            "source_organization": meta.get("sourceOrganization"),
            "source_note": meta.get("sourceNote"),
        }

    completed = datetime.now(timezone.utc)
    manifest = {
        "manifest_schema_version": 1, "source_id": SOURCE_KEY,
        "source_name": "World Bank World Development Indicators",
        "download_id": DATASET_KEY, "dataset_id": DATASET_KEY,
        "release": snapshot_id, "snapshot_id": snapshot_id,
        "release_kind": "rolling_api_snapshot", "release_discovery_mode": "api_capture",
        "source_url": API, "final_url": API, "api_documentation_url": API_DOCS_URL,
        "terms_url": TERMS_URL,
        "attribution": "World Bank Group, World Development Indicators; originating organisations and source notes retained per indicator.",
        "license_status": "unclear_or_unverified_indicator_level_third_party_review_required",
        "retrieved_at": completed.isoformat(), "capture_started_at": started.isoformat(),
        "snapshot_scope": {
            "profile": "wp1_5_area_and_density", "work_package": "WP1.5",
            "field_ids": [3, 33, 34], "world_bank_source_id": SOURCE_ID,
            "start_year": start_year, "end_year": end_year,
            "primary_indicators": list(INDICATORS), "all_indicators": list(INDICATORS),
            "api_version": "v2", "observation_api_per_page": PER_PAGE,
            "country_catalogue_record_count": len(country_rows),
            "country_catalogue_page_count": country_pages,
            "observation_status": "Preserved verbatim; not interpreted as actual, estimate or forecast.",
        },
        "indicator_summaries": summaries, "files": artifacts,
    }
    common._write_json(manifest_path, manifest)
    report = {
        "assessment": "world_bank_wdi_wp1_5_area_immutable_api_snapshot",
        "snapshot_id": snapshot_id, "retrieved_at": completed.isoformat(),
        "manifest_path": manifest_path.relative_to(root).as_posix(),
        "artifact_count": len(artifacts), "artifact_bytes": sum(x["byte_count"] for x in artifacts),
        "country_catalogue_record_count": len(country_rows),
        "period": {"start_year": start_year, "end_year": end_year},
        "indicators": summaries,
        "source_last_updated": source_payload[0].get("lastupdated"),
        "side_effects": {"raw_snapshot_acquired": True, "manifest_written": True,
            "project_database_written": False, "game_database_written": False,
            "candidate_report_written": False, "approval_created": False,
            "source_selection_changed": False},
    }
    report_dir = root / REPORT_BASE / snapshot_id
    common._write_json(report_dir / "acquisition.json", report)
    lines = ["# WDI WP1.5 area and density snapshot", "", f"- Snapshot: {snapshot_id}",
        f"- Period: {start_year}–{end_year}", f"- Manifest: {report['manifest_path']}",
        f"- Raw response artifacts: {len(artifacts)} ({report['artifact_bytes']:,} bytes)", "",
        "| Field | Indicator | Meaning | Rows | API last updated |",
        "|---:|---|---|---:|---|"]
    for code, spec in INDICATORS.items():
        item = summaries[code]
        lines.append(f"| {spec['field_id']} | {code} | {spec['meaning']} | {item['observation_rows']} | {item['api_last_updated'] or 'not supplied'} |")
    lines.extend(["", "Raw response bytes, URLs, headers, retrieval times, byte counts and SHA-256 hashes are preserved.",
        "This is a rolling API snapshot, not a provider-declared release vintage.",

def _read(root: Path, artifact: dict[str, Any]) -> Any:
    rel = artifact.get("local_raw_path")
    if not isinstance(rel, str):
        raise AreaSnapshotError("Manifest artifact has no raw path")
    path = (root / rel).resolve(strict=True)
    path.relative_to(root)
    raw = path.read_bytes()
    if len(raw) != artifact.get("byte_count") or hashlib.sha256(raw).hexdigest() != artifact.get("sha256"):
        raise AreaSnapshotError(f"Snapshot integrity mismatch: {rel}")
    try:
        return json.loads(raw)
    except json.JSONDecodeError as exc:
        raise AreaSnapshotError(f"Invalid JSON artifact: {rel}") from exc


def load_snapshot(manifest_path: str | Path, *, repository_root: str | Path = ROOT) -> dict[str, Any]:
    root = Path(repository_root).resolve()
    path = Path(manifest_path)
    if not path.is_absolute():
        path = root / path
    path = path.resolve(strict=True)
    path.relative_to(root)
    manifest = json.loads(path.read_text(encoding="utf-8"))
    if manifest.get("source_id") != SOURCE_KEY or manifest.get("download_id") != DATASET_KEY:
        raise AreaSnapshotError("Manifest is not the WP1.5 area/density WDI dataset")
    scope = manifest.get("snapshot_scope") or {}
    if scope.get("profile") != "wp1_5_area_and_density" or scope.get("field_ids") != [3, 33, 34]:
        raise AreaSnapshotError("Unexpected WP1.5 WDI snapshot scope")
    artifacts = manifest.get("files")
    if not isinstance(artifacts, list) or not artifacts:
        raise AreaSnapshotError("Manifest contains no artifacts")
    source = None
    countries: dict[str, dict[str, Any]] = {}
    metadata = {}
    pages: dict[str, list[tuple[int, int, list[dict[str, Any]]]]] = {}
    country_page_records: list[tuple[int, int]] = []
    for item in artifacts:
        name = Path(item["local_raw_path"]).name
        payload = _read(root, item)
        if name == "source_metadata.json":
            source = payload
        elif name.startswith("country_metadata_page_"):
            page_number = int(item.get("page_number") or 0)
            pages_count = int(item.get("page_count") or 0)
            country_page_records.append((page_number, pages_count))
            for row in payload[1]:
                iso3 = row.get("id")
                if isinstance(iso3, str):
                    if iso3 in countries:
                        raise AreaSnapshotError(f"Duplicate country catalogue ISO3: {iso3}")
                    countries[iso3] = row
        elif name.startswith("indicator_metadata_"):
            code = name.removeprefix("indicator_metadata_").removesuffix(".json")
            row = next((r for r in payload[1] if r.get("id") == code), None)
            if row is None or str((row.get("source") or {}).get("id")) != SOURCE_ID:
                raise AreaSnapshotError(f"Indicator metadata mismatch: {code}")
            metadata[code] = row
        elif name.startswith("observations_"):
            stem = name.removeprefix("observations_").removesuffix(".json")
            code, page_text = stem.rsplit("_page_", 1)
            pages.setdefault(code, []).append((int(page_text), int(item.get("page_count") or 0), payload[1]))
    if source is None or not any(str(x.get("id")) == SOURCE_ID for x in source[1]):
        raise AreaSnapshotError("Snapshot lacks WDI source metadata")
    country_page_records.sort()
    expected_country_pages = country_page_records[0][1] if country_page_records else 0
    if expected_country_pages != len(country_page_records) or [x[0] for x in country_page_records] != list(range(1, expected_country_pages + 1)) or any(x[1] != expected_country_pages for x in country_page_records):
        raise AreaSnapshotError("Country metadata pagination is incomplete")
    if set(metadata) != set(INDICATORS) or set(pages) != set(INDICATORS):
        raise AreaSnapshotError("Snapshot does not contain all three indicator bundles")
    observations = {}
    for code, bundle in pages.items():
        bundle.sort(key=lambda x: x[0])
        expected = bundle[0][1]
        if expected != len(bundle) or [x[0] for x in bundle] != list(range(1, expected + 1)) or any(x[1] != expected for x in bundle):
            raise AreaSnapshotError(f"{code}: incomplete observation pagination")
        observations[code] = [row for _, _, rows in bundle for row in rows]
    return {"manifest": manifest, "manifest_path": path,
        "manifest_sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
        "countries": countries, "indicator_metadata": metadata, "observations": observations}
